#include <tunables/global>

# Modelled on the official example profile: capabilities and network are restricted, file access is
# broad with the host mappings denied. Hand-listed file rules broke s6 twice (0.3.0: exec without read
# on /init; 0.3.1: no directory listing of /etc/services.d) and cannot be validated without a Supervisor.

profile ha_harmony_editor flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities: s6 init (chown /data, drop to the app user) and nothing else.
  capability chown,
  capability fowner,
  capability setuid,
  capability setgid,
  capability kill,
  capability dac_override,
  capability dac_read_search,
  capability fsetid,
  capability setpcap,

  signal (send) set=(kill,term,int,hup,cont),

  # Network: IP sockets only (no raw or packet sockets).
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix,
  deny network raw,
  deny network packet,

  # Files: broad, like the official example profile. A hand-listed set silently broke the start
  # chain: s6's services-up lists the DIRECTORY /etc/services.d (`/etc/services.d/**` does not match
  # the directory itself), got nothing under AppArmor, and so never started the editor service (0.3.1:
  # no "copying legacy longrun editor" line, no app output). Host mappings are denied below; deny wins.
  file,
  ptrace (trace,read) peer=docker-default,
  signal (receive) set=(kill,term) peer=unconfined,
  signal (send,receive) set=(kill,term) peer=docker-default,

  # Host mappings: denied explicitly (none are mapped either)
  deny /config/** rwklx,
  deny /homeassistant/** rwklx,
  deny /share/** rwklx,
  deny /ssl/** rwklx,
  deny /media/** rwklx,
  deny /backup/** rwklx,
  deny /addons/** rwklx,
  deny ptrace,
  deny mount,
}
