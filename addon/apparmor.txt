#include <tunables/global>

# Least-privilege profile for the editor add-on, modelled on the official example profile.
# Every executable also needs READ: /init and the s6 service scripts are #!/bin/sh scripts, and
# the interpreter (busybox sh) must open them. "ix" alone lets the kernel exec them but made sh fail
# with "can't open '/init': Permission denied" (0.3.0). Not yet loaded under a real Supervisor.

profile ha_harmony_editor flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities: s6 init (chown /data, drop to the app user) and nothing else.
  capability chown,
  capability fowner,
  capability setuid,
  capability setgid,
  capability kill,
  capability dac_override,
  capability dac_read_search,
  capability fsetid,
  capability setpcap,

  signal (send) set=(kill,term,int,hup,cont),

  # Network: IP sockets only (no raw or packet sockets).
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix,
  deny network raw,
  deny network packet,

  # S6-Overlay (base image init)
  /init rix,
  /bin/** rix,
  /sbin/** rix,
  /usr/bin/** rix,
  /usr/sbin/** rix,
  /usr/lib/bashio/** rix,
  /run/{s6,s6-rc*,service}/** rix,
  /package/** rix,
  /command/** rix,
  /etc/s6-overlay/** r,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /etc/fix-attrs.d/** r,
  /run/{,**} rwk,
  /dev/tty rw,
  /dev/null rw,
  /dev/{,**} r,
  /proc/** r,
  /sys/fs/cgroup/** r,
  /dev/urandom r,

  # Python runtime and the application (read-only)
  /usr/local/** mrix,
  /usr/lib/** mr,
  /lib/** mr,
  /opt/app/** r,
  /opt/deps/** mr,
  /etc/{passwd,group,nsswitch.conf,hosts,resolv.conf,ssl/**,mime.types} r,
  /proc/sys/net/core/somaxconn r,

  # The add-on's own data and scratch space
  /data/ r,
  /data/** rwk,
  /tmp/** rwk,

  # Host mappings: denied explicitly (none are mapped either)
  deny /config/** rwklx,
  deny /homeassistant/** rwklx,
  deny /share/** rwklx,
  deny /ssl/** rwklx,
  deny /media/** rwklx,
  deny /backup/** rwklx,
  deny /addons/** rwklx,
  deny ptrace,
  deny mount,
}
