#include <tunables/global>

# Least-privilege profile for the editor add-on. NOT yet exercised under a Supervisor: loading it
# and tuning it in complain mode is still to do (see the add-on security guide).
profile ha_harmony_editor flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities: s6 init (chown /data, drop to the app user) and nothing else.
  capability chown,
  capability fowner,
  capability setuid,
  capability setgid,
  capability kill,
  capability dac_override,

  signal (send) set=(kill,term,int,hup,cont),

  # Network: IP sockets only (no raw or packet sockets).
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix,
  deny network raw,
  deny network packet,

  # S6-Overlay (base image init)
  /init ix,
  /bin/** ix,
  /sbin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  /dev/null rw,
  /dev/urandom r,

  # Python runtime and the application (read-only)
  /usr/local/** rix,
  /usr/lib/** mr,
  /lib/** mr,
  /opt/app/** r,
  /opt/deps/** mr,
  /etc/{passwd,group,nsswitch.conf,hosts,resolv.conf,ssl/**,mime.types} r,
  /proc/sys/net/core/somaxconn r,

  # The add-on's own data and scratch space
  /data/ r,
  /data/** rwk,
  /tmp/** rwk,

  # Host mappings: denied explicitly (none are mapped either)
  deny /config/** rwklx,
  deny /homeassistant/** rwklx,
  deny /share/** rwklx,
  deny /ssl/** rwklx,
  deny /media/** rwklx,
  deny /backup/** rwklx,
  deny /addons/** rwklx,
  deny ptrace,
  deny mount,
}
